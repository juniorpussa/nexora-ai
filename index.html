export async function onRequestPost(context) { const { request, env } = context; try { if (!env.GEMINI_API_KEY) { return jsonResponse({ error: "GEMINI_API_KEY não configurada na Cloudflare." }, 500); } const body = await request.json(); const { mode = "image", imageBase64, mimeType = "image/jpeg", prompt = "" } = body; if (!imageBase64) { return jsonResponse({ error: "Envie a foto do produto antes de gerar." }, 400); } // ========================================== // MODO 1: GERAÇÃO DE IMAGEM COMERCIAL // ========================================== if (mode === "image") { const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-image:generateContent?key=${env.GEMINI_API_KEY}`; const payload = { contents: [ { parts: [ { inline_data: { mime_type: mimeType, data: imageBase64 } }, { text: prompt } ] } ], generationConfig: { responseModalities: ["IMAGE"] } }; const geminiRes = await fetch(apiUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); const data = await geminiRes.json(); if (!geminiRes.ok) { const msg = data?.error?.message || `Erro na API Gemini (HTTP ${geminiRes.status})`; return jsonResponse({ error: msg }, geminiRes.status); } const parts = data?.candidates?.[0]?.content?.parts || []; const imagePart = parts.find((p) => p.inlineData || p.inline_data); const inline = imagePart?.inlineData || imagePart?.inline_data; if (!inline?.data) { return jsonResponse({ error: "A IA não retornou nenhuma imagem. Tente outro estilo." }, 502); } return jsonResponse({ imageBase64: inline.data, mimeType: inline.mimeType || inline.mime_type || "image/png" }); } // ========================================== // MODO 2: GERAÇÃO DE VÍDEO DINÂMICO (VEO) // ========================================== if (mode === "video") { const videoModel = "veo-3.1-fast-generate-preview"; const startUrl = `https://generativelanguage.googleapis.com/v1beta/models/${videoModel}:predictLongRunning?key=${env.GEMINI_API_KEY}`; const startPayload = { instances: [ { prompt: prompt, image: { bytesBase64Encoded: imageBase64, mimeType: mimeType } } ], parameters: { aspectRatio: "9:16", personGeneration: "dont_allow" } }; const startRes = await fetch(startUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(startPayload) }); const startData = await startRes.json(); if (!startRes.ok) { const msg = startData?.error?.message || `Erro ao iniciar vídeo (HTTP ${startRes.status})`; return jsonResponse({ error: msg }, startRes.status); } const operationName = startData?.name; if (!operationName) { return jsonResponse({ error: "Operação de vídeo não retornou identificador." }, 502); } // Aguarda conclusão (polling até ~55s dentro do limite da Function) const opUrl = `https://generativelanguage.googleapis.com/v1beta/${operationName}?key=${env.GEMINI_API_KEY}`; let opData = startData; for (let i = 0; i < 14; i++) { if (opData.done) break; await new Promise((r) => setTimeout(r, 4000)); const pollRes = await fetch(opUrl); opData = await pollRes.json(); } if (!opData.done) { return jsonResponse({ error: "O vídeo ainda está sendo renderizado no servidor. Tente novamente em alguns segundos." }, 504); } if (opData.error) { return jsonResponse({ error: opData.error.message || "Falha na renderização do vídeo." }, 502); } const samples = opData?.response?.generateVideoResponse?.generatedSamples || opData?.response?.generatedVideos || []; const videoUri = samples[0]?.video?.uri || samples[0]?.uri; if (!videoUri) { return jsonResponse({ error: "O servidor concluiu sem retornar o arquivo de vídeo." }, 502); } const separator = videoUri.includes("?") ? "&" : "?"; const fileRes = await fetch(`${videoUri}${separator}key=${env.GEMINI_API_KEY}`); if (!fileRes.ok) { return jsonResponse({ error: `Falha ao baixar o vídeo gerado (HTTP ${fileRes.status}).` }, 502); } const arrayBuffer = await fileRes.arrayBuffer(); const base64Video = arrayBufferToBase64(arrayBuffer); return jsonResponse({ videoBase64: base64Video, mimeType: "video/mp4" }); } return jsonResponse({ error: "Modo inválido." }, 400); } catch (err) { return jsonResponse({ error: err.message || "Erro interno no servidor." }, 500); } } function arrayBufferToBase64(buffer) { let binary = ""; const bytes = new Uint8Array(buffer); const chunkSize = 0x8000; for (let i = 0; i < bytes.length; i += chunkSize) { binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunkSize)); } return btoa(binary); } function jsonResponse(obj, status = 200) { return new Response(JSON.stringify(obj), { status, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); }
